#include <Servo.h>

// Define Pin Numbers based on your screenshots
const int redLedPin = 2;     // Red LED (+) -> D2
const int yellowLedPin = 3;  // Yellow LED (+) -> D3
const int greenLedPin = 4;   // Green LED (+) -> D4
const int irSensorPin = 7;   // IR Sensor OUT -> D7
const int buzzerPin = 8;     // Buzzer Positive (+) -> D8
const int servoPin = 9;      // Servo Orange/Yellow Wire -> D9

Servo barrierServo;          // Create servo object

void setup() {
  Serial.begin(9600);

  // Initialize pin modes
  pinMode(redLedPin, OUTPUT);
  pinMode(yellowLedPin, OUTPUT);
  pinMode(greenLedPin, OUTPUT);
  pinMode(buzzerPin, OUTPUT);
  pinMode(irSensorPin, INPUT);

  // Attach servo to pin 9
  barrierServo.attach(servoPin);

  // Default state: Clear everything out on startup
  barrierServo.write(90); // Start closed
  digitalWrite(redLedPin, LOW);
  digitalWrite(yellowLedPin, LOW);
  digitalWrite(greenLedPin, LOW);
  digitalWrite(buzzerPin, LOW);
}

void loop() {
  // --- PHASE 1: RED LIGHT ON (BARRIER CLOSED) ---
  Serial.println("Red Light ON - Barrier Closed");
  digitalWrite(redLedPin, HIGH);   // Turn Red LED ON
  digitalWrite(greenLedPin, LOW);  // Ensure Green LED is OFF
  digitalWrite(yellowLedPin, LOW); // Ensure Yellow LED is OFF
  barrierServo.write(90);          // Close the barrier (90 degrees)

  // Continuously scan for an object during the 5-second window
  for (int i = 0; i < 50; i++) { 
    int sensorState = digitalRead(irSensorPin);
    
    // Most IR sensors output LOW when an obstacle is detected
    if (sensorState == LOW) { 
      Serial.println("Object Detected! Triggering Buzzer.");
      digitalWrite(buzzerPin, HIGH); // Turn buzzer ON
      delay(5000);                   // Beep for 5 seconds
      digitalWrite(buzzerPin, LOW);  // Turn buzzer OFF
      break;                         // Exit scanning loop early
    }
    delay(100);                      // 50 iterations * 100ms = 5 seconds total window
  }
  
  digitalWrite(redLedPin, LOW);    // Turn Red LED OFF after time window expires
  delay(200);                      // Small transition delay


  // --- PHASE 2: GREEN LIGHT ON (BARRIER OPEN) ---
  Serial.println("Green Light ON - Barrier Open");
  digitalWrite(greenLedPin, HIGH); // Turn Green LED ON
  digitalWrite(redLedPin, LOW);    // Ensure Red LED is OFF
  digitalWrite(yellowLedPin, LOW); // Ensure Yellow LED is OFF
  digitalWrite(buzzerPin, LOW);    // Explicitly enforce buzzer off
  barrierServo.write(180);         // Open the barrier (180 degrees)
  
  delay(5000);                     // Keep it open for 5 seconds
  
  digitalWrite(greenLedPin, LOW);  // Turn Green LED OFF after 5 seconds
  delay(200);                      // Small transition delay before looping back
}
